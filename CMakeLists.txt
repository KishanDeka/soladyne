cmake_minimum_required(VERSION 3.18)
project(solar_dynamo_cpp VERSION 0.2.0 LANGUAGES CXX)

option(SOLAR_DYNAMO_BUILD_PYTHON "Build Python extension" ON)
option(SOLAR_DYNAMO_BUILD_TESTS "Build native tests" OFF)

add_library(solar_dynamo_core STATIC cpp/src/model.cpp)
target_include_directories(solar_dynamo_core PUBLIC cpp/include)
target_compile_features(solar_dynamo_core PUBLIC cxx_std_17)
set_target_properties(solar_dynamo_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(solar_dynamo_core PRIVATE /W4)
else()
  target_compile_options(solar_dynamo_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(SOLAR_DYNAMO_BUILD_PYTHON)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/src/bindings.cpp)
  target_link_libraries(_core PRIVATE solar_dynamo_core)
  install(TARGETS _core LIBRARY DESTINATION solar_dynamo)
endif()

if(SOLAR_DYNAMO_BUILD_TESTS)
  enable_testing()
  add_executable(test_model tests/cpp/test_model.cpp)
  target_link_libraries(test_model PRIVATE solar_dynamo_core)
  add_test(NAME native_model COMMAND test_model)
endif()
